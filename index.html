ESP32 / ARDUINO PCA9685 DRIVER SKETCH

Flash this sketch directly to your ESP32, Arduino Uno, or Teensy. It drives the PCA9685 16-channel I2C servo shield and listens on Serial (115200 baud) for commands streamed directly from this web simulation! 

#include <Wire.h>
#include <Adafruit_PWMServoDriver.h>

// Default PCA9685 I2C address is 0x40
Adafruit_PWMServoDriver pwm = Adafruit_PWMServoDriver(0x40);

#define SERVO_FREQ 50  // 50Hz standard analog/digital servos
#define USMIN  500     // 0 degrees (500us pulse)
#define USMAX  2500    // 180 degrees (2500us pulse)

// Map 0-180 degrees to PCA9685 12-bit tick count (0-4095)
void setServoAngle(uint8_t channel, float angle) {
  angle = constrain(angle, 0.0, 180.0);
  uint16_t pulseLength = map(angle * 10, 0, 1800, USMIN, USMAX);
  uint16_t pulselength_ticks = (pulseLength * 4096) / 20000;
  pwm.setPWM(channel, 0, pulselength_ticks);
}

void setup() {
  Serial.begin(115200);
  Wire.begin();
  pwm.begin();
  pwm.setPWMFreq(SERVO_FREQ);
  delay(500);

  // Initialize all joints to safe 90 deg neutral
  for(int ch = 0; ch < 16; ch++) {
    setServoAngle(ch, 90.0);
    delay(50);
  }
  Serial.println("MAKERBOT_HARDWARE_READY");
}

void loop() {
  // Protocol: "S:<channel>:<angle>\n"
  if (Serial.available()) {
    String msg = Serial.readStringUntil('\n');
    if (msg.startsWith("S:")) {
      int firstColon = msg.indexOf(':');
      int secondColon = msg.indexOf(':', firstColon + 1);
      int ch = msg.substring(firstColon + 1, secondColon).toInt();
      float ang = msg.substring(secondColon + 1).toFloat();
      setServoAngle(ch, ang);
    }
  }
}
